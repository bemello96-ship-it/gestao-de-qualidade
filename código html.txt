<script>
  const STORAGE_KEY = 'tarefas_qa_app';

  function obterTarefas() {
    return JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];
  }

  function salvarTarefas(tarefas) {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(tarefas));
  }

  function validarDados(titulo, responsavel, vencimento) {
    if (!titulo || titulo.trim() === '') return 'Erro: O título da tarefa é obrigatório.';
    if (!responsavel || responsavel.trim() === '') return 'Erro: O responsável é obrigatório.';
    if (!vencimento) return 'Erro: A data de vencimento é obrigatória.';
    return '';
  }

  function renderizarTabela() {
    const tbody = document.getElementById('listaTarefas');
    const tarefas = obterTarefas();
    tbody.innerHTML = '';

    if (tarefas.length === 0) {
      tbody.innerHTML = '<tr><td colspan="6">Nenhuma tarefa cadastrada.</td></tr>';
      return;
    }

    tarefas.forEach(t => {
      const tr = document.createElement('tr');
      tr.innerHTML = `
        <td>${t.titulo}</td>
        <td>${t.descricao || '-'}</td>
        <td>${t.responsavel}</td>
        <td>${t.vencimento}</td>
        <td>
          <select onchange="alterarStatus('${t.id}', this.value)" aria-label="Status de ${t.titulo}">
            <option value="Pendente" ${t.status === 'Pendente' ? 'selected' : ''}>Pendente</option>
            <option value="Em andamento" ${t.status === 'Em andamento' ? 'selected' : ''}>Em andamento</option>
            <option value="Concluída" ${t.status === 'Concluída' ? 'selected' : ''}>Concluída</option>
          </select>
        </td>
        <td>
          <button class="btn-edit" onclick="editarTarefa('${t.id}')">Editar</button>
          <button class="btn-delete" onclick="excluirTarefa('${t.id}')">Excluir</button>
        </td>
      `;
      tbody.appendChild(tr);
    });
  }

  document.getElementById('formTarefa').addEventListener('submit', function(e) {
    e.preventDefault();
    const id = document.getElementById('tarefaId').value;
    const titulo = document.getElementById('titulo').value;
    const descricao = document.getElementById('descricao').value;
    const responsavel = document.getElementById('responsavel').value;
    const vencimento = document.getElementById('vencimento').value;
    const status = document.getElementById('status').value;

    const erro = validarDados(titulo, responsavel, vencimento);
    const divErro = document.getElementById('mensagemErro');
    if (erro) {
      divErro.textContent = erro;
      return;
    }
    divErro.textContent = '';

    let tarefas = obterTarefas();
    if (id) {
      tarefas = tarefas.map(t => t.id === id ? { id, titulo: titulo.trim(), descricao: descricao.trim(), responsavel: responsavel.trim(), vencimento, status } : t);
    } else {
      tarefas.push({
        id: Date.now().toString(),
        titulo: titulo.trim(),
        descricao: descricao.trim(),
        responsavel: responsavel.trim(),
        vencimento,
        status
      });
    }

    salvarTarefas(tarefas);
    limparFormulario();
    renderizarTabela();
  });

  function editarTarefa(id) {
    const tarefa = obterTarefas().find(t => t.id === id);
    if (!tarefa) return;
    document.getElementById('tarefaId').value = tarefa.id;
    document.getElementById('titulo').value = tarefa.titulo;
    document.getElementById('descricao').value = tarefa.descricao;
    document.getElementById('responsavel').value = tarefa.responsavel;
    document.getElementById('vencimento').value = tarefa.vencimento;
    document.getElementById('status').value = tarefa.status;
    document.getElementById('btnSalvar').textContent = 'Atualizar Tarefa';
    document.getElementById('btnCancelar').style.display = 'inline-block';
  }

  function alterarStatus(id, novoStatus) {
    const tarefas = obterTarefas().map(t => t.id === id ? { ...t, status: novoStatus } : t);
    salvarTarefas(tarefas);
  }

  function excluirTarefa(id) {
    const tarefas = obterTarefas().filter(t => t.id !== id);
    salvarTarefas(tarefas);
    renderizarTabela();
  }

  function limparFormulario() {
    document.getElementById('formTarefa').reset();
    document.getElementById('tarefaId').value = '';
    document.getElementById('btnSalvar').textContent = 'Salvar Tarefa';
    document.getElementById('btnCancelar').style.display = 'none';
    document.getElementById('mensagemErro').textContent = '';
  }

  document.getElementById('btnCancelar').addEventListener('click', limparFormulario);

  // SUÍTE DE TESTES AUTOMATIZADOS (Caixa Branca + Integração com LocalStorage)
  function executarTestesAutomatizados() {
    const backup = localStorage.getItem(STORAGE_KEY);
    localStorage.removeItem(STORAGE_KEY);
    const logs = [];
    let passou = 0, falhou = 0;

    function assert(condicao, nomeTeste) {
      if (condicao) {
        passou++;
        logs.push(`[PASS] ${nomeTeste}`);
      } else {
        falhou++;
        logs.push(`[FAIL] ${nomeTeste}`);
      }
    }

    // Teste 1: Validação de campos obrigatórios (Caixa Branca na função validarDados)
    assert(validarDados('', 'João', '2026-12-01') !== '', 'T01 - Bloquear tarefa sem título');
    assert(validarDados('Estudar QA', '', '2026-12-01') !== '', 'T02 - Bloquear tarefa sem responsável');
    assert(validarDados('Estudar QA', 'João', '2026-12-01') === '', 'T03 - Aceitar dados válidos');

    // Teste 2: Persistência e alteração de status no LocalStorage
    salvarTarefas([{ id: '1', titulo: 'Teste Auto', descricao: '', responsavel: 'QA', vencimento: '2026-12-01', status: 'Pendente' }]);
    alterarStatus('1', 'Concluída');
    const recarregada = obterTarefas().find(t => t.id === '1');
    assert(recarregada && recarregada.status === 'Concluída', 'T04 - Persistir alteração de status no LocalStorage');

    // Teste 3: Exclusão
    excluirTarefa('1');
    assert(obterTarefas().length === 0, 'T05 - Excluir tarefa e atualizar armazenamento');

    // Restaura dados originais do usuário
    if (backup) localStorage.setItem(STORAGE_KEY, backup);
    else localStorage.removeItem(STORAGE_KEY);
    renderizarTabela();

    const resumo = `=== EXECUÇÃO AUTOMATIZADA ===\nTotal: ${passou + falhou} | Aprovados: ${passou} | Reprovados: ${falhou}\n\n` + logs.join('\n');
    console.log(resumo);
    const pre = document.getElementById('saidaTeste');
    pre.style.display = 'block';
    pre.textContent = resumo;
  }

  renderizarTabela();
</script>
